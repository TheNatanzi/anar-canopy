# Anar Events website — council rubric (9 seats)

You are one seat on a nine-seat council auditing a static website for Anar Events, a wedding planning and coordination studio for Persian, Afghan and South Asian families in the San Francisco Bay Area. Three models (Codex Astra, Claude Fable, Gemini Argon) each hold three seats:
- **Wedding planner** (knows Persian, Afghan and South Asian weddings and the families who pay): does the site win trust and inquiries from couples AND parents? Is the cultural fluency real, respectful, not costume? Is the path to "Inquire" obvious?
- **Brand creative lead**: does it feel like the Anar brand book (night palette first: cypress #1B221E, parchment #F9F6F0, gold #C99E5C, ruby #6D1A24; TT Ramillas Bold; the plain gold logo never retyped; the kashida line; the painted pomegranate tree that frames the edges and bottom but never covers the middle)? Premium? Alive with motion but never gimmicky? No grid hairlines.
- **Web developer**: code quality, motion implementation (GSAP/ScrollTrigger/Lenis), performance (lazy loading, no layout shift, video weight), mobile at 390px, accessibility (landmarks, focus, reduced motion, contrast), no console errors, no horizontal overflow.

Hard rules the site must obey (score ≤6 if broken): nothing about catering, decor or rentals; the hero shows the plain gold logo (no web address) readable over the video; the tree art frames edges and bottom only, middle open; English only (Persian accents allowed); TT Ramillas Bold for Latin text; night palette first.

Score 0–10. 9 = "I would sign off for launch with trivial nits". 10 = world class. Be harsh and specific.

Reply ONLY with JSON: {"model":"<astra|fable|argon>","role":"<planner|brand|dev>","score":<number>,"blocking":["..."],"fixes":[{"where":"file:line or section","what":"exact change","why":"one line"}]} with at most 6 fixes, most important first.

===== assets/js/site.js =====
/* =========================================================
   Anar Events — site.js (v2, night palette, 3 Nov 2026)
   Loaded (defer, pinned) before this file:
     GSAP 3.12.5 + ScrollTrigger (cdnjs), Lenis 1.1.18 (unpkg)
   Every motion is a numbered block below. If GSAP is missing or
   prefers-reduced-motion is on, html gets .reduced: nothing is
   hidden and nothing scrubs. Nothing here is page-specific; each
   block checks for its own elements.
   ========================================================= */
(function () {
  'use strict';

  var html = document.documentElement;
  var body = document.body;
  var reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var hasGsap = typeof window.gsap !== 'undefined' && typeof window.ScrollTrigger !== 'undefined';
  var motion = hasGsap && !reduced;
  var isDesktop = function () { return window.matchMedia('(min-width: 768px)').matches; };
  var vh = function () { return window.innerHeight; };
  var $ = function (s, c) { return (c || document).querySelector(s); };
  var $$ = function (s, c) { return Array.prototype.slice.call((c || document).querySelectorAll(s)); };

  html.classList.add(motion ? 'motion' : 'reduced');
  if (motion) { gsap.registerPlugin(ScrollTrigger); }

  /* ---------------------------------------------------------
     0. Hero video source: 1080 on wide screens, 720 at <=768px.
        (A <source media> attribute is ignored inside <video>, so
        the file is chosen here.) Reduced motion: poster only.
     --------------------------------------------------------- */
  var heroVideo = $('.hero__video');
  if (heroVideo && !reduced) {
    var src = isDesktop() ? heroVideo.getAttribute('data-src-desktop') : heroVideo.getAttribute('data-src-mobile');
    if (src) { heroVideo.src = src; heroVideo.load(); heroVideo.play().catch(function () {}); }
    if ('IntersectionObserver' in window) {
      new IntersectionObserver(function (entries) {
        entries.forEach(function (en) {
          if (en.isIntersecting) { heroVideo.play().catch(function () {}); } else { heroVideo.pause(); }
        });
      }, { threshold: 0.05 }).observe(heroVideo);
    }
  }

  /* ---------------------------------------------------------
     Smooth scroll (Lenis) wired to the GSAP ticker
     --------------------------------------------------------- */
  var lenis = null;
  if (motion && typeof window.Lenis !== 'undefined') {
    lenis = new Lenis({ lerp: 0.1, smoothWheel: true });
    lenis.on('scroll', ScrollTrigger.update);
    gsap.ticker.add(function (t) { lenis.raf(t * 1000); });
    gsap.ticker.lagSmoothing(0);
    html.classList.add('lenis');
  }
  function headerOffset() { var h = $('.site-header'); return -(h ? h.offsetHeight : 0); }
  function scrollToEl(el) {
    if (lenis) { lenis.scrollTo(el, { offset: headerOffset(), duration: 1.4 }); }
    else { el.scrollIntoView({ behavior: reduced ? 'auto' : 'smooth', block: 'start' }); }
  }
  /* Same-page anchors (including "index.html#inquire" while on index) */
  $$('a[href*="#"]').forEach(function (a) {
    a.addEventListener('click', function (e) {
      var url = new URL(a.getAttribute('href'), location.href);
      var norm = function (p) { return p.replace(/\/index\.html$/, '/'); };
      if (norm(url.pathname) !== norm(location.pathname) || !url.hash) { return; }
      var target = document.getElementById(url.hash.slice(1));
      if (!target) { return; }
      e.preventDefault();
      closeNav();
      scrollToEl(target);
      history.pushState(null, '', url.hash);
    });
  });
  if (location.hash) {
    var landing = document.getElementById(location.hash.slice(1));
    if (landing) { setTimeout(function () { scrollToEl(landing); }, 80); }
  }

  /* ---------------------------------------------------------
     Header: solid once scrolled; mobile drawer
     --------------------------------------------------------- */
  var header = $('.site-header');
  var navToggle = $('.nav-toggle');
  function onScrollState() {
    if (!header) { return; }
    header.classList.toggle('is-scrolled', (window.scrollY || window.pageYOffset) > 40);
  }
  if (lenis) { lenis.on('scroll', onScrollState); } else { window.addEventListener('scroll', onScrollState, { passive: true }); }
  onScrollState();
  function closeNav() {
    if (!header || !navToggle) { return; }
    header.classList.remove('nav-open');
    navToggle.setAttribute('aria-expanded', 'false');
  }
  if (navToggle) {
    navToggle.addEventListener('click', function () {
      var open = header.classList.toggle('nav-open');
      navToggle.setAttribute('aria-expanded', open ? 'true' : 'false');
    });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape') { closeNav(); } });
  }

  /* ---------------------------------------------------------
     "How we work +" slide-out panel (works without GSAP too):
     gold tab, night panel, close ×, Esc closes, backdrop closes,
     basic focus trap, focus returns to the tab.
     --------------------------------------------------------- */
  var tab = $('.hww-tab'), panel = $('#hww-panel'), backdrop = $('.hww-backdrop');
  if (tab && panel && backdrop) {
    var closeBtn = $('.hww-panel__close', panel);
    var lastFocus = null;
    var focusables = function () {
      return $$('a[href], button:not([disabled]), input, select, textarea, [tabindex]:not([tabindex="-1"])', panel);
    };
    var openPanel = function () {
      lastFocus = document.activeElement;
      panel.hidden = false; backdrop.hidden = false;
      requestAnimationFrame(function () { panel.classList.add('is-open'); backdrop.classList.add('is-on'); });
      tab.setAttribute('aria-expanded', 'true');
      body.classList.add('panel-open');
      if (lenis) { lenis.stop(); }
      setTimeout(function () { (closeBtn || panel).focus(); }, 80);
    };
    var closePanel = function () {
      panel.classList.remove('is-open'); backdrop.classList.remove('is-on');
      tab.setAttribute('aria-expanded', 'false');
      body.classList.remove('panel-open');
      if (lenis) { lenis.start(); }
      setTimeout(function () { panel.hidden = true; backdrop.hidden = true; }, reduced ? 0 : 560);
      if (lastFocus && lastFocus.focus) { lastFocus.focus(); }
    };
    tab.addEventListener('click', openPanel);
    if (closeBtn) { closeBtn.addEventListener('click', closePanel); }
    backdrop.addEventListener('click', closePanel);
    document.addEventListener('keydown', function (e) {
      if (panel.hidden) { return; }
      if (e.key === 'Escape') { closePanel(); return; }
      if (e.key === 'Tab') {
        var f = focusables(); if (!f.length) { return; }
        var first = f[0], last = f[f.length - 1];
        if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
        else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
      }
    });
    $$('a[href]', panel).forEach(function (a) { a.addEventListener('click', function () { closePanel(); }); });
  }

  /* ---------------------------------------------------------
     Accordion (The celebrations): hover expands on pointer
     devices, click/focus expands (touch, keyboard). Stacked
     cards on mobile (CSS), so nothing to do there.
     --------------------------------------------------------- */
  $$('[data-accordion]').forEach(function (acc) {
    var panels = $$('.accordion__panel', acc);
    var hoverable = window.matchMedia('(hover: hover)').matches;
    function open(target) {
      panels.forEach(function (p) {
        var on = p === target;
        p.classList.toggle('is-open', on);
        var btn = $('.accordion__trigger', p);
        if (btn) { btn.setAttribute('aria-expanded', on ? 'true' : 'false'); }
      });
    }
    panels.forEach(function (p) {
      var btn = $('.accordion__trigger', p);
      if (hoverable) { p.addEventListener('mouseenter', function () { if (isDesktop()) { open(p); } }); }
      if (btn) {
        btn.addEventListener('click', function () { open(p); });
        btn.addEventListener('focus', function () { if (isDesktop()) { open(p); } });
      }
    });
  });

  if (!motion) { return; }   /* ===== everything below is motion only ===== */

  /* ---------------------------------------------------------
     1. HERO (scrubbed over the hero's own height):
        the logo blurs 0 -> 10px and rises slower than the page;
        the video tint deepens to solid cypress;
     2. the canopy-u3 piece moves up and out at 0.6x scroll speed
        (the section moves at 1x, so the art gets +0.4x back).
     --------------------------------------------------------- */
  var hero = $('.hero');
  if (hero) {
    gsap.timeline({
      scrollTrigger: { trigger: hero, start: 'top top', end: 'bottom top', scrub: true, invalidateOnRefresh: true }
    })
      .to('.hero__mark', { y: function () { return vh() * 0.45; }, filter: 'blur(10px)', opacity: 0.35, ease: 'none', duration: 1 }, 0)
      .to('.hero__canopy', { y: function () { return vh() * 0.4; }, ease: 'none', duration: 1 }, 0)
      .to('.hero__deepen', { opacity: 1, ease: 'none', duration: 1 }, 0)
      .to(['.hero__scroll', '.hero__tag'], { opacity: 0, ease: 'none', duration: 0.35 }, 0);
  }

  /* ---------------------------------------------------------
     3. DESCEND THE TREE: trunks-tall.png sits in a sticky,
        viewport-high frame and scrolls past at 0.8x while the
        statement and the pillars (normal flow) pass at 1x.
     --------------------------------------------------------- */
  var descend = $('.descend');
  var trunksImg = $('.descend__trunks img');
  if (descend && trunksImg) {
    gsap.to(trunksImg, {
      y: function () {
        var travel = Math.max(0, descend.offsetHeight - vh()) * 0.8;
        var room = Math.max(0, trunksImg.offsetHeight - vh());
        return -Math.min(travel, room);
      },
      ease: 'none',
      scrollTrigger: { trigger: descend, start: 'top top', end: 'bottom bottom', scrub: true, invalidateOnRefresh: true }
    });
  }

  /* ---------------------------------------------------------
     4. Kashida draw-on: every .kashida-draw path draws with
        scroll (stroke-dashoffset from full length to 0)
     --------------------------------------------------------- */
  $$('.kashida-draw path').forEach(function (p) {
    var len = p.getTotalLength();
    p.style.strokeDasharray = len;
    p.style.strokeDashoffset = len;
    gsap.to(p, {
      strokeDashoffset: 0, ease: 'none',
      scrollTrigger: { trigger: p.closest('svg'), start: 'top 88%', end: 'top 45%', scrub: true }
    });
  });

  /* ---------------------------------------------------------
     5. The journey (and the steps line on How we work):
        [data-track] holds .track-line path(s) and .track-dot(s).
        ONE gold line draws across the section; each dot pops as
        the line reaches it.
     --------------------------------------------------------- */
  $$('[data-track]').forEach(function (track) {
    var paths = $$('.track-line path', track);
    var dots = $$('.track-dot', track);
    if (!paths.length) { return; }
    var tl = gsap.timeline({
      scrollTrigger: { trigger: track, start: 'top 78%', end: 'bottom 60%', scrub: true }
    });
    paths.forEach(function (p) {
      var len = p.getTotalLength();
      p.style.strokeDasharray = len;
      p.style.strokeDashoffset = len;
      tl.to(p, { strokeDashoffset: 0, ease: 'none', duration: 1 }, 0);
    });
    dots.forEach(function (d, i) {
      tl.fromTo(d, { scale: 0 }, { scale: 1, ease: 'back.out(2)', duration: 0.06 }, Math.max(0, (i + 0.5) / dots.length - 0.04));
    });
  });

  /* ---------------------------------------------------------
     6. How we work teaser: three gold kashida strips slide in
        from the right as the section arrives, then the cards.
     --------------------------------------------------------- */
  var teaser = $('.teaser');
  if (teaser) {
    var strips = $$('.teaser__strips i', teaser);
    gsap.timeline({ scrollTrigger: { trigger: teaser, start: 'top 75%', end: 'top 20%', scrub: true } })
      .fromTo(strips, { xPercent: 102 }, { xPercent: 0, ease: 'power1.inOut', duration: 0.6, stagger: 0.15 }, 0);
  }

  /* ---------------------------------------------------------
     7. The roots: roots-floor.png rises in as the floor while
        the Inquire section arrives.
     --------------------------------------------------------- */
  var ground = $('.ground'), roots = $('.ground__roots');
  if (ground && roots) {
    gsap.fromTo(roots, { yPercent: 28 }, {
      yPercent: 0, ease: 'none',
      scrollTrigger: { trigger: ground, start: 'top 90%', end: 'top 15%', scrub: true }
    });
  }

  /* ---------------------------------------------------------
     8. Floating leaves drift slowly (two places). Reduced
        motion never reaches this block, so they stay still.
     --------------------------------------------------------- */
  $$('.leaf').forEach(function (leaf, i) {
    gsap.to(leaf, { y: 18 + i * 6, x: i % 2 ? -10 : 10, rotation: i % 2 ? -3 : 3, duration: 6 + i * 1.5, ease: 'sine.inOut', yoyo: true, repeat: -1 });
    gsap.to(leaf, { y: '+=' + (40 + i * 10), ease: 'none', scrollTrigger: { trigger: leaf.parentElement, start: 'top bottom', end: 'bottom top', scrub: true } });
  });

  /* ---------------------------------------------------------
     9. Section reveals (fade + rise, 0.9s ease-out), once;
        images scale 1.02 -> 1 on reveal; .stagger children stagger.
     --------------------------------------------------------- */
  $$('.reveal').forEach(function (el) {
    gsap.to(el, {
      opacity: 1, y: 0, duration: 0.9, ease: 'power2.out',
      scrollTrigger: { trigger: el, start: 'top 88%', once: true }
    });
  });
  $$('.reveal-img').forEach(function (el) {
    gsap.to(el, { scale: 1, duration: 1.2, ease: 'power2.out', scrollTrigger: { trigger: el, start: 'top 88%', once: true } });
  });
  $$('.stagger').forEach(function (group) {
    gsap.to(group.children, {
      opacity: 1, y: 0, duration: 0.8, ease: 'power2.out', stagger: 0.08,
      scrollTrigger: { trigger: group, start: 'top 88%', once: true }
    });
  });

  /* Refresh once fonts and images have settled (sticky heights, line lengths) */
  window.addEventListener('load', function () { ScrollTrigger.refresh(); });
  if (document.fonts && document.fonts.ready) { document.fonts.ready.then(function () { ScrollTrigger.refresh(); }); }
  if (trunksImg && !trunksImg.complete) { trunksImg.addEventListener('load', function () { ScrollTrigger.refresh(); }); }
})();
